# 什么是堆栈

## 计算机如何进行表达式求值

对于算术表达式，如`5 + 6 / 2 - 3 * 4`，
- 由两类对象构成
	- 运算数，如2，3，4
	- 运算符号，如+
- 不同运算符号优先级不一样

## 后缀表达式

- 中缀表达式：运算符号位于两个运算数==之间==。如`a + b * c - d / e`
- 后缀表达式：运算符号位于两个运算数==之后==。如`abc * + de / -`
	(前缀表达式：运算符号位于两个运算数==之前==。如`- + a *bc /de`)

后缀表达式求值策略：从左向右“扫描”，逐个处理运算数和运算符号
- 遇到运算数：顺序存储运算数
- 遇到运算符号：倒序输出运算结果

启示：==需要有种存储方法，顺序存储运算数，并在需要时倒序输出==

## 堆栈的抽象数据类型描述

==堆栈(stack)==：具有一定操作约束的线性表，只在一端(栈顶,Top)做插入、删除
- 插入数据：入栈(Push)
-  删除数据：出栈(Pop)
- ==后入先出原则==：Last in first out(LIFO)

- 类型名称：堆栈(stack)
- 数据对象集：一个有0个或多个元素的有穷线性表
- 操作集：长度为`Maxsize`的堆栈$S\in \mathrm{Stack}$，堆栈元素$\mathrm{item}\in \mathrm{ElemenType}$
	- `Stack CreateStack(int Maxsize)`生成空堆栈，其最大长度为`Maxsize`
	- `int IsFull(Stack S, int Maxsize)`判断堆栈$S$是否已满
	- `void Push(Stack S, ElemenType item)`将元素`item`压入堆栈
	- `int IsEmpty(Stack S)`判断堆栈$S$是否为空
	- `ElemenType Pop(Stack S)`删除并返回弹出的栈顶元素

# 堆栈的顺序存储实现

栈的顺序存储结构通常由一个==一维数组==和一个记录==栈顶==元素位置的变量组成

~~~
typedef struct SNode{

    ElementType data[Maxsize];

    int top;

} Snode;

int IsFull(Snode* s){

    return s->top == Maxsize - 1;

}

int IsEmpty(Snode* s){

    return s->top == -1;

}

void Push(Snode* s, ElementType x){

    if (IsFull(s)){

        printf("Stack is full.\n");

        return;

    }

    s->data[++(s->top)] = x;

    return;

}

ElementType Pop(Snode* s){

    if(IsEmpty(s)){

        printf("Stack is empty.\n");

        return;

    }

    return s->data[(s->top)--];

}
~~~

例：如何用一个数组实现两个堆栈，并最大程度利用数组空间，使得数组只要有空间，入栈操作就可以成功

==以数组的两端分别作为两个堆栈的栈底==。当两个栈的栈顶指针相遇时，表示两个栈都满了。

~~~
typedef struct DNode{

    ElementType data[Maxsize];

    int top1;

    int top2;

} Dnode;

int IsFull(Dnode* s){

    return (s->top2) - (s->top1) == 1;

}

int Stack1IsEmpty(Dnode* s){

    return (s->top1 == -1);

}

int Stack2IsEmpty(Dnode* s){

    return (s->top2 == Maxsize);

}

void Push(Dnode* s, ElementType x, int Tag){

    if(IsFull(s)){

        printf("Stack is full.\n");

        return;

    }

    if(Tag == 1){

        s->data[++(s->top1)] = x;

    }

    else{

        s->data[--(s->top2)] = x;

    }

}

ElementType Pop(Dnode* s, int Tag){

    if(Tag == 1){

        if(Stack1IsEmpty(s)){

            printf("Stack1 is empty.\n");

            return;

        }

        return s->data[(s->top1)--];

    }

    else{

        if(Stack2IsEmpty(s)){

            printf("Stack2 is empty.\n");

            return;

        }

        return s->data[(s->top2)++];

    }

}
~~~

# 堆栈的链式存储实现

堆栈的链式存储结构实际上就是一个==单链表==，叫做==链栈==。插入和删除操作只能在链栈的栈顶进行。
且栈顶指针应该在链表的==头节点==。
（单向链表访问只能从头部开始，且只能单向遍历）

无额外头节点的实现方式：
```
#define ERROR -1

typedef int Elementype;

typedef struct Snode{

    Elementype data;

    struct Snode* next;

} Snode;

typedef Snode *Stack;

int IsEmpty(Stack top){

    return (top == NULL);

}

Stack CreateStack(){

    return NULL;

}

void PushNode(Stack* top, Elementype item){

    Stack newNode = (Stack)malloc(sizeof(Snode));

    if (newNode == NULL){

        printf("Fail to allocate the memory.\n");

        return;

    }

    newNode->data = item;

    newNode->next = *top;

    *top = newNode;

    return;

}

Elementype PopNode(Stack* top){

    if (IsEmpty(*top)){

        printf("The stack is empty.\n");

        return ERROR;

    }

    Stack tmp = *top;

    *top = (*top)->next;

    Elementype tmpdata = tmp->data;

    free(tmp);

    return tmpdata;

}

void PrintStack(Stack top){

    Stack itr = top;

    while(itr != NULL){

        printf("%d\n", itr->data);

        itr = itr->next;

    }

    return;

}
```

有额外头节点的实现方式：
```
#define ERROR -1

typedef int Elementype;

typedef struct Snode{

    Elementype data;

    struct Snode* next;

} Snode;

typedef Snode *Stack;

int IsEmpty(Stack top)

{

    return (top->next == NULL);

}

Stack CreateStack()

{

    Stack top = (Stack)malloc(sizeof(Snode));

    top->next = NULL;

    return top;

}

void PushNode(Stack top, Elementype item)

{

    Stack newNode = (Stack)malloc(sizeof(Snode));

    if (newNode == NULL)

    {

        printf("Fail to allocate the memory.\n");

        return;

    }

    newNode->data = item;

    newNode->next = top->next;

    top->next = newNode;

    return;

}

Elementype PopNode(Stack top)

{

    if (IsEmpty(top))

    {

        printf("The stack is empty.\n");

        return ERROR;

    }

    Stack tmp = top->next;

    top->next = top->next->next;

    Elementype tmpdata = tmp->data;

    free(tmp);

    return tmpdata;

}

void PrintStack(Stack top)

{

    Stack itr = top->next;

    while (itr != NULL)

    {

        printf("%d\n", itr->data);

        itr = itr->next;

    }

    return;

}
```
# 堆栈应用：表达式求值

从左到右读入==后缀表达式==的各项。
对==中缀表达式==求值，将中缀表达式转换为后缀表达式。

## 中缀表达式如何转换为后缀表达式

从头到尾读取==中缀表达式的每个对象==，对不同对象按不同情况处理
- 运算数：直接输出
- 左括号：压入堆栈
- 右括号：将栈顶的运算符弹出并输出，直到遇到左括号（出栈，不输出）
- 运算符：
	- 当前优先级==大于==栈顶运算符时，压栈
	- 当前优先级==小于等于==栈顶运算符时，将栈顶运算符弹出并输出，直到该运算符大于栈顶运算符时将该运算符压栈
- 处理完毕后，将堆栈中运算符全部输出

## 堆栈的其他应用

- 函数调用及递归实现
- 深度优先搜索(DFS)
- 回溯算法
