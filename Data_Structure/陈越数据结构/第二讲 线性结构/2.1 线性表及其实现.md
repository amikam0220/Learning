# 多项式表示

一元多项式及其运算：
$f(x)=a_{0}+a_{1}x+\cdots+a_{n-1}x^{n-1}+a_{n}x^{n}$.
主要运算有多项式相加、相乘

- 如何表示多项式
	关键数据：多项式系数$n$，各项系数$a_{i}$以及指数$i$。

## 方法1：顺序存储结构直接表示

数组下标对应多项式的次数，各分量对应其系数
两个多项式相加可以将对应分量直接相加
- 优点：
	运算方便
- 问题：
	可能存在次数相差过大，中间存在过多无效数据0

## 方法2：顺序存储结构表示非零项

每个数组分量对应两个信息：系数和次数（用结构数组表示），每个分量对应一个非零项
相加过程：从头开始，比较两个多项式当前对应项的指数，若不等则输出较大项，若相等则将系数相加再输出

```
struct polynomial* polyaddition(struct polynomial a[],struct polynomial b[], int sizeofa, int sizeofb){

    const int sizeofc = sizeofa + sizeofb;

    struct polynomial* c = (struct polynomial*)malloc(sizeofc * sizeof(struct polynomial));

    int i = 0, j = 0, k = 0;

    while (i < sizeofa && j < sizeofb){

        if (a[i].exp == b[j].exp){

            c[k].coe = a[i].coe + b[j].coe;

            c[k].exp = a[i].exp;

            i++, j++, k++;

        }else if (a[i].exp > b[j].exp){

            c[k].coe = a[i].coe;

            c[k].exp = a[i].exp;

            i++, k++;

        }else{

            c[k].coe = b[j].coe;

            c[k].exp = b[j].exp;

            j++, k++;

        }

    }

    return c;

}
```

# 线性表及其顺序存储

## 什么是线性表（Linear List）

由同类型==数据元素==构成==有序序列==的线性结构
- 表中元素个数成为线性表的==长度==
- 没有元素时成为==空表==
- 起始位置称==表头==，结束位置称==表尾==

## 线性表的抽象数据类型描述

- 类型名称：
	线性表(List)
- 数据对象集：
	线性表是元素构成的有序序列($a_{1},a_{2},\cdots,a_{n}$)
- 操作集：
	线性表$L\in \mathrm{List}$，整数$i$表示位置，元素$X\in \mathrm{Elementype}$，基本操作有：
	- `List MakeEmpty()`：初始化一个空表
	- `ElemenType FindKth(int K, List L)`：根据位序K返回相应元素
	- `int Find(ElemenType X, List L)`：找到元素X在L中第一次出现的位置并返回
	- `void insert(ElemenType X, int i, List L)`：在位序i前插入一个新元素X
	- `void Delete (int i, List L)`：删除位序i处的元素
	- `int Length(List L)`：返回线性表L的长度n

## 线性表的顺序存储实现

利用==数组==的==连续存储空间顺序==存放线性表的各元素
~~~
typedef struct LNode *List;

struct LNode{

    Elementype data[MaxSize];

    int last;//the last position of the array

};
~~~
~~~
List MakeEmpty(){

    List L = (List)malloc(sizeof(struct LNode));

    L->last = -1;

    return L;

}

Elementype FindKth(int K, List L){

    if (K <= L->last && K >= 0)

        return L->data[K];

    else

        return -1;

}

int Find(Elementype X, List L){

    int k = -1;

    int left = 0, right = L->last, mid = (right - left) / 2 + left;

    while (left < right){

        if (L->data[mid] == X){

            k = mid;

            break;

        }else if(L->data[mid] < X){

            left = mid + 1;

        }else{

            right = mid;

        }

    }

    return k;

}

void insert(Elementype X, int i, List L){

    if (L->last + 1 >= MaxSize){

        printf("FULL.\n");

        return;

    }

    if (i < 0 || i > L->last + 1){

        printf("Error Position.\n");

        return;

    }

    for (int j = L->last; j >= i; j--){

        L->data[j + 1] = L->data[j];

    }

    L->data[i] = X, (L->last)++;

    return;

}

void delete(int i, List L){

    if (i < 0 || i > L->last){

        printf("Error Position.\n");

        return;

    }

    for (int j = i; j < L->last; j++){

        L->data[j] = L->data[j + 1];

    }

    (L->last)--;

}

int Length(List L){

    return (L->last) + 1;

}
~~~

# 链式存储及查找

## 线性表的链式存储实现

不要求==逻辑上相邻==的元素物理上也相邻，通过“链”建立逻辑关系，操作时不需要移动数据，只需要修改链。
```
typedef struct Lnode *List;

typedef struct Lnode{

    Elementype data;

    List Next;

}Lnode;

List MakeEmpty(){

    List head = (List)malloc(sizeof(Lnode));

    head->Next = NULL;

    return head;

}

List FindKth(int k, List L){

    List itr = L;

    int i = 0;

    for (i = 0; i < k; i++){

        itr = itr->Next;

    }

    if(i == k){

        return itr;

    }else{

        return NULL;

    }

}

List Find(Elementype X, List L){

    List itr = L;

    int cnt = -1;

    for (; itr->Next != NULL; itr = itr->Next){

        cnt++;

        if (itr->data == X){

            break;

        }

    }

    return (cnt == -1)? NULL : itr;

}

List Insert(Elementype X, int i, List L){

    if (i == 0){

        List newNode = (List)malloc(sizeof(Lnode));

        newNode->data = X;

        newNode->Next = L;

        return newNode;

    }//insert at the beginning

    List itr = L;

    for (int j = 0; j < i - 1; j++){

        itr = itr->Next;

    }

    List newNode = (List)malloc(sizeof(Lnode));

    newNode->data = X;

    newNode->Next = itr->Next;

    itr->Next = newNode;

    return L;

}

List Delete(int i, List L){

    if (i == 0){

        if(L->Next == NULL){

            return NULL;

        }

        List tmp = L->Next;

        free(L);

        L = tmp;

        return L;

    }//delete at the beginning

    List itr = L;

    for (int j = 0; j < i - 1; j++){

        itr = itr->Next;

    }

    List tmp = itr->Next;

    itr->Next = tmp->Next;

    free(tmp);

    return;

}

int Length(List L){

    int cnt = 0;

    List itr = L;

    for (; itr->Next != NULL; itr = itr->Next){

        cnt++;

    }

    return cnt;

}
```

# 广义表与多重链表

例：对于二元多项式$P(x,y)$，如何表示？
可以将二元多项式看成关于$x$的一元多项式，用链表表示
![[广义表figure1.png]]
这种表称为==广义表==。

## 广义表

- 广义表是线性表的推广
- 对于线性表而言，$n$个元素都是基本的==单元素==
- 广义表中，这些元素不仅可以是单元素也可以是另一个==广义表==。
一般而言通过设置==标志域==(Tag)来区分该节点是单元素还是另一个广义表，同时将子表指针域$\mathrm{Sublist}$与单元素的数据域$\mathrm{Data}$复用(即通过`union`共用存储空间)。
则广义表分为标志域、数据域、指针域。
![[广义表figure2.png]]
~~~
typedef struct Gnode *Glist;

struct Gnode{

    int Tag;//tag for different type of node, 0 for data node, 1 for sublist node

    union{

        int data;

        Glist sublist;

    }Uregion;//union region to store data or sublist node

    Glist next;//pointer to next node in the list

};
~~~

## 多重链表

链表中的节点可能同时隶属于多个链。
- ==指针域会有多个==，如上面例子中的`next`和`sublist`两个指针域
- 包含两个指针域的链表==不一定==是多重链表，如双向链表不是多重链表
==树、图==等相对复杂的数据结构都可以采用==多重链表==方式实现存储。

例：用==十字链表==存储稀疏矩阵
- 只存储矩阵的非零元素项
	节点的==数据域==：行坐标(Row)、列坐标(Col)、数值(Value)
- 每个节点通过==两个指针域==把同行、同列串起来：
	- 行指针（向右指针）：Right
	- 列指针（向下指针）：Down
用标识域区分头节点和非零元素节点，头节点的标识值为Head，矩阵非零元素节点的标识值为Term。
每行、每列通过头节点进行遍历，并通过循环链表实现。每个Term节点既属于某一行又属于某一列，即两个链表。整个稀疏矩阵的入口也为Term节点，存储行数、列数以及非零项个数并指向各行各列的头节点。

![[多重链表figure1.png]]

~~~
typedef struct Matrix* Mlist;

struct Matrix{

    int Tag;

    union{

        struct data{

            int row, col, val;

        };

        Mlist next;

    }Uregion;

    Mlist right;

    Mlist down;

};
~~~

下为十字链表的具体实现（此处行、列头节点的具体实现并非课程中的同一个节点而是独立的节点）
```
#include <stdio.h>

#include <stdlib.h>

typedef struct Mnode *Mlist;

typedef struct Mnode

{

    int Tag;

    union

    {

        struct data

        {

            int row, col, val;

        } data;

        Mlist next;

    } Uregion;

    Mlist right;

    Mlist down;

} Mnode;

typedef struct Matrix

{

    Mlist Head;

    Mlist *rowhead;

    Mlist *colhead;

} Matrix;

Mlist createNode(int Tag)

{

    Mlist node = (Mlist)malloc(sizeof(struct Mnode));

    if (!node)

    {

        printf("Fail to allocate the memory.\n");

        exit(1);

    }

    node->Tag = Tag;

    node->right = node;

    node->down = node;

    return node;

}

Matrix *createMatrix(int rows, int cols, int nums)

{

    Matrix *mat = (Matrix *)malloc(sizeof(Matrix));//should be sizeof Matrix

    mat->Head = createNode(1);

    mat->Head->Uregion.data.row = rows;

    mat->Head->Uregion.data.col = cols;

    mat->Head->Uregion.data.val = nums;

    mat->rowhead = (Mlist* )malloc(rows * sizeof(Mlist));//should be sizeof Mlist

    for (int i = 0; i < rows; i++)

    {

        mat->rowhead[i] = createNode(0);

    }

    mat->colhead = (Mlist* )malloc(cols * sizeof(Mlist));

    for (int i = 0; i < cols; i++)

    {

        mat->colhead[i] = createNode(0);

    }

    return mat;

}

void insertNode(Matrix *mat, int row, int col, int val)

{

    // check the validity of row and col

    if (row < 0 ||

        row >= mat->Head->Uregion.data.row ||

        col < 0 ||

        col >= mat->Head->Uregion.data.col)

    {

        printf("Error position.\n");

        return;

    }

    // create Term node(Tag = 1)

    Mlist newNode = createNode(1);

    newNode->Uregion.data.row = row;

    newNode->Uregion.data.col = col;

    newNode->Uregion.data.val = val;

    // insert to the list of rows

    Mlist rowHead = mat->rowhead[row];

    Mlist rowCurr = rowHead->right;

    // find the position

    while (rowCurr->right != rowHead && rowCurr->Uregion.data.col < col)

    {

        rowCurr = rowCurr->right;

    }

    // insert

    newNode->right = rowCurr->right;

    rowCurr->right = newNode;

    // insert to the list of col

    Mlist colHead = mat->colhead[col];

    Mlist colCurr = colHead->down;

    // find the position

    while (colCurr->down != colHead && colCurr->Uregion.data.row < row)

    {

        colCurr = colCurr->down;

    }

    // insert

    newNode->down = colCurr->down;

    colCurr->down = newNode;

}

void freeMatrix(Matrix *mat)

{

    int rownum = mat->Head->Uregion.data.row,

        colnum = mat->Head->Uregion.data.col,

        NonZeronum = mat->Head->Uregion.data.val;

    for (int i = 0; i < colnum; i++)

    {

        Mlist colHead = mat->colhead[i];

        Mlist colCurr = colHead->down;

        Mlist coltmp = colCurr;

        while (colCurr != colHead)

        {

            coltmp = colCurr;

            colCurr = colCurr->down;

            free(coltmp);

        }

        free(colHead);

    }

    for (int j = 0; j < rownum; j++)

    {

        Mlist rowHead = mat->rowhead[j];

        rowHead->right = NULL;

        free(rowHead);

    }

    free(mat->Head);

    free(mat->colhead);

    free(mat->rowhead);

    free(mat);

}

void PrintMatrix(Matrix *mat)

{

    printf("The message of the matrix:"

           "%d rows, %d columns, %d nonzeroelements.\n",

           mat->Head->Uregion.data.row,

           mat->Head->Uregion.data.col,

           mat->Head->Uregion.data.val);

    printf("Print the matrix row by row.\n");

    int rownum = mat->Head->Uregion.data.row;

    for (int i = 0; i < rownum; i++)

    {

        printf("Row %d: ", i);

        Mlist rowHead = mat->rowhead[i];

        Mlist rowCurr = rowHead->right;

        while (rowCurr != rowHead)

        {

            printf("(%d,%d,%d)",

                   rowCurr->Uregion.data.row,

                   rowCurr->Uregion.data.col,

                   rowCurr->Uregion.data.val);

            rowCurr = rowCurr->right;

        }

        printf("\n");

    }

}

int main()

{

    Matrix *mat = createMatrix(4, 5, 7);

    insertNode(mat, 0, 0, 18);

    insertNode(mat, 0, 3, 2);

    insertNode(mat, 1, 1, 27);

    insertNode(mat, 2, 3, -4);

    insertNode(mat, 3, 0, 23);

    insertNode(mat, 3, 1, -1);

    insertNode(mat, 3, 4, 12);

    PrintMatrix(mat);

    freeMatrix(mat);

    return 0;

}
```
