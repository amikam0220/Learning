# 应用实例：多项式加法运算

主要思路：将相同指数的项系数相加，其余部分进行拷贝

```
#include <stdio.h>

#include <stdlib.h>

typedef struct PolyNode{

    int expo;

    int coef;

    struct PolyNode *next;

} Polynode;

typedef Polynode *Polylist;

void AddnewNode(Polylist* poly, int exp, int coe){

    if (coe == 0){

        return;

    }//filter the wrong items

    Polylist newnode = (Polylist)malloc(sizeof(Polynode));

    newnode->coef = coe, newnode->expo = exp;

    newnode->next = NULL;

    if (*poly == NULL){

        *poly = newnode;

        return;

    }else{

        Polylist tmpitr = *poly;

        while(tmpitr->next != NULL){

            tmpitr = tmpitr->next;

        }

        tmpitr->next = newnode;

        return;

    }

}

Polylist Addition(Polylist p1, Polylist p2){

    Polylist itr1 = p1, itr2 = p2;

    Polylist ans = NULL;

    while(itr1 != NULL && itr2 != NULL){

        if (itr1->expo > itr2->expo){

            AddnewNode(&ans, itr1->expo, itr1->coef);

            itr1 = itr1->next;

        }else if (itr1->expo < itr2->expo){

            AddnewNode(&ans, itr2->expo, itr2->coef);

            itr2 = itr2->next;

        }else{

            AddnewNode(&ans, itr1->expo, itr1->coef + itr2->coef);

            itr1 = itr1->next;

            itr2 = itr2->next;

        }

    }

    if (itr1 == NULL){

        while (itr2 != NULL){

            AddnewNode(&ans, itr2->expo, itr2->coef);

            itr2 = itr2->next;

        }

    }else{

        while(itr1 != NULL){

            AddnewNode(&ans, itr1->expo, itr1->coef);

            itr1 = itr1->next;

        }

    }

    return ans;

}

void PrintPolynomial(Polylist poly){

    Polylist itr = poly;

    printf("The result is:\n");

    while (itr != NULL)

    {

        printf("(%d,%d)", itr->expo, itr->coef);

        itr = itr->next;

    }

    return;

}

void FreePolynomial(Polylist poly){

    while(poly != NULL){

        Polylist tmp = poly;

        poly = poly->next;

        free(tmp);

    }

    return;

}

int main(){

    Polylist p1 = NULL, p2 = NULL;

    int tmpexpo = -1;

    int tmpcoef = -1;

    printf("Please enter the exponents "

        "and coefficients of polynomial 1 decreasingly. "

        "Enter zero as coefficient to exit the enter.\n");

    while(tmpcoef != 0){    

        scanf("%d%d", &tmpexpo, &tmpcoef);

        AddnewNode(&p1, tmpexpo, tmpcoef);

    }

    tmpcoef = -1;

    printf("Please enter the exponents "

           "and coefficients of polynomial 2 decreasingly. "

           "Enter zero as coefficient to exit the enter.\n");

    while (tmpcoef != 0){

        scanf("%d%d", &tmpexpo, &tmpcoef);

        AddnewNode(&p2, tmpexpo, tmpcoef);

    }

    Polylist p3 = Addition(p1, p2);

    PrintPolynomial(p3);

    FreePolynomial(p1), FreePolynomial(p2), FreePolynomial(p3);

    return 0;

}
```
注意事项：
- 在`AddnewNode()`函数中，注意检查加入项的系数是否为零以过滤多余项。同时注意传入指针的指针，以便修改外部多项式指针。
- 注意释放多项式。

# 多项式乘法运算

将乘法运算转化为加法运算，即p1的当前项乘p2多项式再加到结果多项式里
然后逐项插入

```
Polylist Multiplication(Polylist p1, Polylist p2){

    if (!p1 || !p2){

        printf("One of the polynomials is empty.\n");

        return NULL;

    }//judge the boundary condition

    Polylist itr1 = p1, itr2 = p2;

    Polylist ans = NULL;

    Polylist tmp = NULL;

    Polylist merged = NULL;

    while (itr1){

        while(itr2){

            tmp = NULL;

            AddnewNode(&tmp, itr1->expo + itr2->expo, itr1->coef * itr2->coef);

            merged = Addition(ans, tmp);//note use variable to receive the result

            FreePolynomial(ans);//free the original memory of ans

            ans = merged;

            FreePolynomial(tmp);//free the memory of tmp to store the current item

            itr2 = itr2->next;

        }

        itr2 = p2;

        itr1 = itr1->next;

    }

    return ans;

}
```
